<objective>
Remove two incorrect filtering rules from pi transcript normalization:
1. Fix 5: Delete the `/^(?:Warning|Note):\s/i` rule from `shouldDropPiChromeLine` — it strips legitimate AI response content. Startup blockers are already caught at launch by `startupBlockers` before normalization runs.
2. Fix 6: Delete `trimmed === '>'` from `looksLikePiSnapshot` — bare `>` is too ambiguous (Bash PS2, heredoc, markdown blockquote) and causes false-positive pi classification.

Purpose: Prevent normalization from silently discarding legitimate AI content from pi responses.
Output: Modified `transcript-normalization.ts` (2 lines removed), updated test suite with corrected assertions.
</objective>

export function looksLikePiSnapshot(lines: string[]) {
  return lines.some((line) => {
    const trimmed = line.trim()
    return (
      trimmed.includes('Welcome to pi') ||
      /^pi\s+v\d+\.\d+\.\d+/i.test(trimmed) ||
      trimmed.includes('Type your message') ||
      trimmed.includes('run /help') ||
      trimmed === '>'        // ← DELETE THIS LINE (Fix 6)
    )
  })
}

function shouldDropPiChromeLine(line: string) {
  const trimmed = line.trim()
  if (!trimmed) {

    trimmed.includes('Welcome to pi') ||
    /^pi\s+v\d+\.\d+\.\d+/i.test(trimmed) ||

  )
}
