    ```typescript
    import { createInterface } from 'node:readline/promises'
    import {
      ensureTTY,
      ensureDaemonNotRunning,
      writeEditableConfigAtomic,
      withWizardCleanup,
      promptMasked,
    } from './setup-wizard-utils.ts'
    import { applyBootstrapBotsToConfig } from '../../config/channels/channel-bot-management.ts'
    import type { ChannelBootstrapBots } from '../../config/channels/channel-bot-management.ts'
    import { getDefaultChannelAvailability } from '../commands/startup-bootstrap.ts'
    import { listStartupChannelDescriptors } from '../../channels/catalog/registry.ts'
    import {
      startDetachedRuntime,
      ensureConfigFile,
    } from '../runtime/runtime-process.ts'
    import { readEditableConfig } from '../../config/core/config-file.ts'
    import {
      expandHomePath,
      getDefaultConfigPath,
      getDefaultRuntimeCredentialsPath,
    } from '../../infra/paths.ts'

    export async function runChannelsWizard(options?: { configPath?: string }): Promise<void> {
      ensureTTY()

      const configPath = expandHomePath(
        options?.configPath ?? process.env.CLISBOT_CONFIG_PATH ?? getDefaultConfigPath(),
      )

      await ensureDaemonNotRunning(configPath)

      await withWizardCleanup(async () => {
        const rl = createInterface({ input: process.stdin, output: process.stdout })

        try {
          await runWizardSession(rl, configPath)
        } finally {
          rl.close()
          process.stdin.unref()
        }
      }, configPath)
    }

    async function runWizardSession(
      rl: Awaited<ReturnType<typeof createInterface>>,
      configPath: string,
    ): Promise<void> {
      console.log('')
      console.log('=== clisbot setup channels ===')
      console.log('')

      const availability = getDefaultChannelAvailability(process.env)
      const collected: ChannelBootstrapBots = {
        telegram: [],
        slack: [],
        'zalo-bot': [],
        'zalo-personal': [],
      }

      for (const descriptor of listStartupChannelDescriptors()) {
        if (descriptor.channel === 'zalo-personal') {
          console.log('')
          console.log(`Zalo Personal: Requires QR login — run: clisbot bots login zalo-personal`)
          continue
        }

        console.log('')
        console.log(descriptor.statusLabel)

        if (availability[descriptor.channel]) {
          console.log(`  Token found in environment variable`)
          collected[descriptor.channel] = [
            { botId: 'default', botToken: { kind: 'env', placeholder: descriptor.statusLabel } },
          ]
          continue
        }

        const answer = await rl.question(`  Configure this channel? [Y/n] `)
        if (answer.trim().toLowerCase() === 'n') {
          continue
        }

        const token = await promptMasked(rl, `  Token: `)
        if (token.trim().length === 0) {
          console.log(`  Skipped (empty input).`)
          continue
        }

        collected[descriptor.channel] = [
          { botId: 'default', botToken: { kind: 'mem', secret: token.trim() } },
        ]
      }

      const hasAny = Object.values(collected).some((bots) => bots.length > 0)
      if (!hasAny) {
        console.log('')
        console.log('No channels configured. Exiting without changes.')
        return
      }

      displayReview(collected)

      const confirmed = await rl.question('Proceed? [Y/n] ')
      if (confirmed.trim().toLowerCase() === 'n') {
        console.log('Setup cancelled.')
        return
      }

      const configResult = await ensureConfigFile(configPath)
      const { config } = await readEditableConfig(configResult.configPath)
      applyBootstrapBotsToConfig(config, collected, { firstRun: false })
      await writeEditableConfigAtomic(configResult.configPath, JSON.stringify(config, null, 2))

      const result = await startDetachedRuntime({
        scriptPath: process.argv[1]!,
        configPath: configResult.configPath,
        runtimeCredentialsPath: getDefaultRuntimeCredentialsPath(),
      })

      console.log('')
      console.log('Channel wizard complete.')
      console.log(`Runtime started (pid: ${result.pid}).`)
      console.log('')
      console.log('Next step: clisbot setup agent')
    }

    function displayReview(collected: ChannelBootstrapBots): void {
      console.log('')
      console.log('='.repeat(50))
      console.log('REVIEW CHANNEL CONFIGURATION')
      console.log('='.repeat(50))
      console.log('')

      for (const [channel, bots] of Object.entries(collected)) {
        if (bots.length === 0) continue
        const source = bots[0]!.botToken?.kind === 'env' ? 'env variable' : 'entered'
        const label = channel.charAt(0).toUpperCase() + channel.slice(1)
        console.log(`  ${label}`)
        console.log(`    Token source: ${source}`)
        console.log(`    DM policy:    pairing`)
        console.log('')
      }
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| operator keyboard → readline buffer | Token input enters buffer; only echo is masked via promptMasked |
| readline buffer → config file | Token value flows from memory into JSON written to disk |
| config file → runtime process | Runtime reads config containing token credentials on startup |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-05-01 | Information Disclosure | Token echo in terminal | mitigate | Uses promptMasked (Plan 06-03) — _writeToOutput monkey-patch shows asterisks. Per D-01, D-02. |
| T-06-05-02 | Information Disclosure | Token in console.log output | mitigate | displayReview() shows "env variable" or "entered" — never the actual token value. Token is only in collected[].botToken.secret which is not printed. |
| T-06-05-03 | Tampering | Partial config write on process kill | mitigate | writeEditableConfigAtomic uses temp file + rename (Phase 5 FOUND-04). Even SIGKILL during write leaves previous config intact. |
| T-06-05-04 | Tampering | Leading/trailing whitespace in token | mitigate | token.trim() applied before storing in ChannelBootstrapBots — prevents accidental whitespace inclusion in credentials. |
| T-06-05-05 | Denial of Service | Ctrl+C during wizard leaves temp file | mitigate | withWizardCleanup SIGINT handler (Phase 5 FOUND-05) removes .tmp file and exits 130. |
| T-06-05-06 | Denial of Service | Process hangs after rl.close() | mitigate | process.stdin.unref() called in finally block — Bun #21189 mitigation. Prevents event loop from stalling after wizard completes. |
